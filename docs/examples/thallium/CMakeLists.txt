# Helper: register an example executable and track it for PCH sharing.
set(_example_targets)
macro(add_example name source)
    add_executable(${name} ${source})
    target_link_libraries(${name} thallium)
    list(APPEND _example_targets ${name})
endmacro()

# Precompile thallium.hpp once (the first target is the donor); all other
# targets reuse the same .gch via REUSE_FROM — they share identical compile
# flags since they all link only against thallium.
list(GET _example_targets 0 _pch_donor)
target_precompile_headers(${_pch_donor} PRIVATE <thallium.hpp>)
foreach(_t IN LISTS _example_targets)
    if(NOT _t STREQUAL _pch_donor)
        target_precompile_headers(${_t} REUSE_FROM ${_pch_donor})
    endif()
endforeach()
